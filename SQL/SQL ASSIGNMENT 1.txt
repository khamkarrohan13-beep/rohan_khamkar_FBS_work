Q3) Create jobs Table with Constraints
Write an SQL query to create a table named jobs including columns job_id, job_title,
min_salary, max_salary. 
Apply constraints:
- job_id must be Primary Key
- job_title cannot be NULL
- max_salary must not exceed 25000 

create table job(
job_id int primary key,
job_title char(20) not null,
min_salary int,
max_salary int check ( max_salary<=25000) );

Q6) Create jobs_default Table with DEFAULT Values
Write an SQL query to create a table named jobs_default including columns job_id, job_title,
min_salary and max_salary.
Set default values:
- job_title default should be blank (“”)
- min_salary default should be 8000
- max_salary default should be NULL

CREATE TABLE jobs_default (
    job_id INT,
    job_title VARCHAR(50) DEFAULT '',
    min_salary INT DEFAULT 8000,
    max_salary INT DEFAULT NULL
);


Q7) Create departments Table (Composite Primary Key)
Create Department table with following structure:
Field Name Type Null Key
department_id DECIMAL(4,0) NO PRI
department_name VARCHAR(30) NO
manager_id DECIMAL(6,0) NO PRI
location_id DECIMAL(4,0) YES

CREATE TABLE departments (
    department_id DECIMAL(4,0) NOT NULL,
    department_name VARCHAR(30) NOT NULL,
    manager_id DECIMAL(6,0) NOT NULL,
    location_id DECIMAL(4,0),
    PRIMARY KEY (department_id, manager_id)
);

Q8) Create employees Table with Normal Foreign Key
Write an SQL query to create a table named employees including columns:
employee_id, first_name, last_name, email, phone_number, hire_date, job_id,
salary, commission, manager_id and department_id.
Column Name Data Type

employee_id INT
first_name VARCHAR(50)
last_name VARCHAR(50)
email VARCHAR(100)
phone_number VARCHAR(20)
hire_date DATE
job_id INT
salary DECIMAL(10,2)
commission DECIMAL(10,2)
manager_id INT
department_id INT

Apply constraints:
- employee_id must be Primary Key
- department_id must be a Foreign Key referencing departments(department_id)

CREATE TABLE employees (
    employee_id INT PRIMARY KEY,
    first_name VARCHAR(50),
    last_name VARCHAR(50),
    email VARCHAR(100),
    phone_number VARCHAR(20),
    hire_date DATE,
    job_id INT,
    salary DECIMAL(10,2),
    commission DECIMAL(10,2),
    manager_id INT,
    department_id INT,
    FOREIGN KEY (department_id) REFERENCES departments(department_id)
);

Q9) Composite Foreign Key Implementation
Step 1: Create department_managers Table
Create a table named department_managers with following structure:
Field Name Data Type Null Key
department_id DECIMAL(4,0) NO PRI
manager_id DECIMAL(6,0) NO PRI
manager_name VARCHAR(50) NO
Composite Primary Key: (department_id, manager_id)
Step 2: Apply Composite Foreign Key in employees Table
Alter the employees table so that the combination (department_id, manager_id)
must exist in department_managers(department_id, manager_id).

CREATE TABLE department_managers (
    department_id DECIMAL(4,0) NOT NULL,
    manager_id DECIMAL(6,0) NOT NULL,
    manager_name VARCHAR(50) NOT NULL,
    PRIMARY KEY (department_id, manager_id)
);

ALTER TABLE employees
ADD CONSTRAINT fk_emp_dept_manager
FOREIGN KEY (department_id, manager_id)
REFERENCES department_managers(department_id, manager_id);
